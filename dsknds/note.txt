    <div class="btn-group">

      <button id="btn-jquery" onclick="loadWithJQuery()">Load via $.getJSON()</button>
    </div>

    <div class="status-badge" id="method-indicator">Method: Native fetch API</div>

    <table>
      <thead>
        <tr>
          <th>ID</th>

          <th>Company</th>
          <th>City</th>

          <td colspan="4">Loading data...</td>

      </tbody>

  <script>
    const API_URL = 'https://jsonplaceholder.typicode.com/users';

    // METHOD 1: Native JavaScript fetch API
    function loadWithFetch() {
      setActiveButton('btn-fetch', 'Method: Native fetch API');
      
      const tableBody = document.getElementById('table-body');
      tableBody.innerHTML = '<tr><td colspan="4">Loading via fetch()...</td></tr>';

      fetch(API_URL)
        .then(response => {
          if (!response.ok) {
            throw new Error(`HTTP Error! Status: ${response.status}`);
          }
          return response.json();
        })
        .then(users => {
          tableBody.innerHTML = ''; // Clear table
          users.forEach(user => {
            const row = document.createElement('tr');
            row.innerHTML = `
              <td>${user.id}</td>
              <td>${user.name}</td>
              <td>${user.company.name}</td>
              <td>${user.address.city}</td>
            `;
            tableBody.appendChild(row);
          });
        })
        .catch(error => {
          console.error('Fetch error:', error);
          tableBody.innerHTML = `<tr><td colspan="4" style="color:red;">Error loading data via fetch()</td></tr>`;
        });
    }

    // METHOD 2: jQuery $.getJSON()
    function loadWithJQuery() {
      setActiveButton('btn-jquery', 'Method: jQuery $.getJSON()');

      const $tableBody = $('#table-body');
      $tableBody.html('<tr><td colspan="4">Loading via $.getJSON()...</td></tr>');

      $.getJSON(API_URL, function(users) {
        $tableBody.empty(); // Clear table
        $.each(users, function(index, user) {
          const row = `
            <tr>
              <td>${user.id}</td>
              <td>${user.name}</td>
              <td>${user.company.name}</td>
              <td>${user.address.city}</td>
            </tr>
          `;
          $tableBody.append(row);
        });
      }).fail(function(jqxhr, textStatus, error) {
        console.error('jQuery error:', textStatus, error);
        $tableBody.html(`<tr><td colspan="4" style="color:red;">Error loading data via jQuery</td></tr>`);
      });
    }

    // UI Helper: Update active button state
    function setActiveButton(activeId, statusText) {
      document.querySelectorAll('.btn-group button').forEach(btn => btn.classList.remove('active'));
      document.getElementById(activeId).classList.add('active');
      document.getElementById('method-indicator').textContent = statusText;
    }

    // Load initial data on startup
    document.addEventListener('DOMContentLoaded', loadWithFetch);
  </script>
